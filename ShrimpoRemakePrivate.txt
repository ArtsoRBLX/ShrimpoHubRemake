local function OnCharacter(char)
    Shared.Character = char
    Shared.HumanoidRootPart = char:WaitForChild("HumanoidRootPart", 5)
    Shared.Humanoid = char:WaitForChild("Humanoid", 5)
end
LocalPlayer.CharacterAdded:Connect(OnCharacter)

    Shared.Character = LocalPlayer.Character
    Shared.HumanoidRootPart = Shared.Character:FindFirstChild("HumanoidRootPart")
    Shared.Humanoid = Shared.Character:FindFirstChildOfClass("Humanoid")
end

local function teleportTo(name, isExact)
    local Character, HumanoidRootPart = Shared.Character, Shared.HumanoidRootPart
    if not Character or not HumanoidRootPart then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        local match = isExact and (obj.Name == name) or obj.Name:lower():find(name:lower())
        if match then
            local part = obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)) or (obj:IsA("BasePart") and obj)
            if part then
                HumanoidRootPart.CFrame = (part:IsA("Model") and part:GetPivot() or part.CFrame) + Vector3.new(0, 3, 0)
                return true
            end
        end
    end
    return false
end
Shared.teleportTo = teleportTo

-- Noclip
local NoclipConnection = nil
local function toggleNoclip(state)
    if state then
        local function applyNoclip()
            local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    pcall(function()
                        for _, conn in ipairs(getconnections(part:GetPropertyChangedSignal("CanQuery"))) do conn:Disconnect() end
                        for _, conn in ipairs(getconnections(part:GetPropertyChangedSignal("CanCollide"))) do conn:Disconnect() end
                        for _, conn in ipairs(getconnections(part:GetPropertyChangedSignal("CanTouch"))) do conn:Disconnect() end
                        for _, conn in ipairs(getconnections(part.Changed)) do conn:Disconnect() end
                    end)
                    part.CanCollide = false
                end
            end
        end
        applyNoclip()
        if not NoclipConnection then
            NoclipConnection = RunService.Stepped:Connect(function()
                local character = LocalPlayer.Character
                if character then
                    for _, part in ipairs(character:GetDescendants()) do
                        if part:IsA("BasePart") then part.CanCollide = false end
                    end
                end
            end)
        end
    else
        if NoclipConnection then NoclipConnection:Disconnect(); NoclipConnection = nil end
        local character = LocalPlayer.Character
        if character then
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

TpWalkGroup:AddToggle("FloatToggle", {
    Text = "Float", Default = false,
    Callback = function(v)
        floatEnabled = v
        local HumanoidRootPart = Shared.HumanoidRootPart
        if floatEnabled then
            if HumanoidRootPart then
                local rp = RaycastParams.new()
                rp.FilterDescendantsInstances = {Shared.Character}
                rp.FilterType = Enum.RaycastFilterType.Exclude
                local rr = workspace:Raycast(HumanoidRootPart.Position, Vector3.new(0, -1000, 0), rp)
                lockedFloatY = rr and (rr.Position.Y + floatHeight) or HumanoidRootPart.Position.Y
            end
            floatConnection = RunService.RenderStepped:Connect(function()
                local c, r = Shared.Character, Shared.HumanoidRootPart
                if c and r and lockedFloatY then
                    r.CFrame = CFrame.new(r.Position.X, lockedFloatY, r.Position.Z) * (r.CFrame - r.Position)
                    r.AssemblyLinearVelocity = Vector3.new(r.AssemblyLinearVelocity.X, 0, r.AssemblyLinearVelocity.Z)
                end
            end)
        else
            if floatConnection then floatConnection:Disconnect(); floatConnection = nil end
            lockedFloatY = nil
        end
    end
})

    Text = "Float Studs", Default = 3.5, Min = 1, Max = 20, Rounding = 1,
